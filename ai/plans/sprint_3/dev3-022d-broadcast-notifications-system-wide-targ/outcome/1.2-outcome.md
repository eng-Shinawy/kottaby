import type { BroadcastAudienceType } from "@/backend/enum/notifications/broadcast-audience-type.enum";
import type { UserRole } from "@/backend/enum/users/user-role.enum";

export interface BroadcastAudienceSelector {
  readonly type: BroadcastAudienceType;

  readonly country?: string | null;   // meaningful only when type is Country
  readonly planId?: number | null;    // meaningful only when type is Plan
}

}

### Downstream consumption rules encoded in the JSDoc (Phase 2 must honor)

- **Coherence matrix (service-enforced, pre-DB, fail-closed):** `All` ⇒ all companions absent; `Role` ⇒ `role` required, `country`/`planId` absent; `Country` ⇒ `country` required (trimmed, ≤100), `role`/`planId` absent; `Plan` ⇒ `planId` required (positive safe int), `role`/`country` absent.
- **Closed shape:** all fields `readonly`, NO index signature, no escape hatch — consumers map field-by-field; `{ ...input }` spread into persistence is documented as prohibited (BOPLA posture).
- **Request-scoped:** the selector is never persisted (rows record resolved recipients); it has no schema-derived counterpart.
- **`BroadcastNotificationSubmitInput`** maps field-by-field into the engine's batch emit: `title` (trim, ≤255), `body` (`null`-allowed, verbatim, never per-recipient localized), `audience` (recipients resolved server-side only — the shape carries NO identity/recipient field).
